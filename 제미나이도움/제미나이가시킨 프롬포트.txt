우리는 K리그 공식 데이터(kleague.com)를 수집하여 1~100 스케일 능력치로 환산하고, 이를 Cloudflare Workers + D1(SQLite) 환경에서 서비스하는 "K리그 감독 시뮬레이션 게임" 백엔드를 구축하고 있어.

[제공된 코드 및 데이터 구조 설명]
1. `kleague_collector_and_transformer.py`:
   - kleague.com 웹사이트/API에서 선수 프로필 및 시즌 통계(골, 슈팅, 패스, 태클, 공중볼 등)를 수집.
   - 포지션별 가중치 산출 공식 및 Min-Max 구간 보정 로직을 거쳐 1~100 사이의 모든 정수(91~99 포함)로 게임 능력치를 자동 환산.
   - 10명의 K리그 샘플 데이터 수집/변환 예시 포함.
2. `schema.sql`:
   - Cloudflare D1 용 SQL DDL. 모든 능력치 컬럼에 `INTEGER CHECK (value BETWEEN 1 AND 100)` 제약조건 적용.
3. `worker.ts`:
   - Cloudflare Workers REST API (선수 목록/검색, FM 프로필 스타일 상세 조회).

아래 작성된 파이썬 수집/변환 스크립트와 D1 스키마, Workers API 코드를 바탕으로 프로젝트 구조를 세팅하고 D1 데이터베이스 초기화 및 API 엔드포인트를 구축해줘.

import math
import random
import json

def convert_stat_to_100_scale(raw_value, min_raw, max_raw, target_min=45, target_max=99):
    """
    실제 기록 수치를 1~100 스케일로 정밀 매핑하는 함수 (91~99 등 전 구간 보정)
    """
    if max_raw == min_raw:
        normalized = 0.5
    else:
        normalized = (raw_value - min_raw) / (max_raw - min_raw)
        normalized = max(0.0, min(1.0, normalized))
    
    # 기본 선형 매핑
    base_score = target_min + (normalized * (target_max - target_min))
    
    # 5의 배수 편향 방지를 위한 세부 보정 (+-2 오프셋)
    offset = random.choice([-2, -1, 0, 1, 2])
    final_score = math.floor(base_score + offset)
    
    return max(1, min(100, final_score))

def calculate_player_attributes(player_raw_data):
    """
    K리그 실제 기록 기반 포지션별 1~100 능력치 가중치 산출 엔진
    """
    pos = player_raw_data.get("position", "MF")
    stats = player_raw_data.get("stats", {})
    
    # 기본 경기 기록 데이터
    goals = stats.get("goals", 0)
    assists = stats.get("assists", 0)
    shots = stats.get("shots", 0)
    passes = stats.get("passes", 0)
    tackles = stats.get("tackles", 0)
    aerials = stats.get("aerials", 0)
    appearances = max(1, stats.get("appearances", 1))

    # 경기당 평균 기록
    gpg = goals / appearances
    apg = assists / appearances
    spg = shots / appearances
    ppg = passes / appearances
    tpg = tackles / appearances

    # 포지션별 가중치 계산 (1~100)
    if "FW" in pos or "ST" in pos:
        finishing = convert_stat_to_100_scale(gpg, 0.0, 0.8, 50, 98)
        shooting = convert_stat_to_100_scale(spg, 0.5, 4.0, 50, 96)
        passing = convert_stat_to_100_scale(apg, 0.0, 0.4, 45, 88)
        tackling = convert_stat_to_100_scale(tpg, 0.0, 1.0, 30, 65)
        strength = convert_stat_to_100_scale(aerials / appearances, 0.0, 5.0, 45, 92)
    elif "DF" in pos or "CB" in pos:
        finishing = convert_stat_to_100_scale(gpg, 0.0, 0.2, 30, 70)
        shooting = convert_stat_to_100_scale(spg, 0.0, 1.0, 30, 68)
        passing = convert_stat_to_100_scale(ppg, 10.0, 60.0, 50, 88)
        tackling = convert_stat_to_100_scale(tpg, 0.5, 4.0, 55, 97)
        strength = convert_stat_to_100_scale(aerials / appearances, 1.0, 7.0, 60, 96)
    else: # MF
        finishing = convert_stat_to_100_scale(gpg, 0.0, 0.4, 40, 85)
        shooting = convert_stat_to_100_scale(spg, 0.2, 2.5, 45, 88)
        passing = convert_stat_to_100_scale(ppg, 20.0, 80.0, 60, 98)
        tackling = convert_stat_to_100_scale(tpg, 0.2, 3.0, 40, 85)
        strength = convert_stat_to_100_scale(aerials / appearances, 0.2, 4.0, 40, 85)

    return {
        "id": player_raw_data["id"],
        "name_ko": player_raw_data["name_ko"],
        "name_en": player_raw_data["name_en"],
        "position_main": pos,
        "team": player_raw_data["team"],
        "current_ability": math.floor((finishing + shooting + passing + tackling + strength) / 5),
        "attributes": {
            "technical": {
                "finishing": finishing,
                "passing": passing,
                "tackling": tackling,
                "technique": math.floor((passing + shooting) / 2)
            },
            "mental": {
                "vision": math.floor((passing * 0.7) + (assists * 3)),
                "composure": math.floor((finishing * 0.6) + 35)
            },
            "physical": {
                "strength": strength,
                "stamina": convert_stat_to_100_scale(appearances, 5, 38, 50, 95)
            }
        }
    }

DROP TABLE IF EXISTS players;

CREATE TABLE players (
    id TEXT PRIMARY KEY,
    name_ko TEXT NOT NULL,
    name_en TEXT NOT NULL,
    position_main TEXT NOT NULL,
    team TEXT NOT NULL,
    current_ability INTEGER CHECK (current_ability BETWEEN 1 AND 100),
    
    -- Technical Attributes (1~100)
    finishing INTEGER CHECK (finishing BETWEEN 1 AND 100),
    passing INTEGER CHECK (passing BETWEEN 1 AND 100),
    tackling INTEGER CHECK (tackling BETWEEN 1 AND 100),
    technique INTEGER CHECK (technique BETWEEN 1 AND 100),
    
    -- Mental Attributes (1~100)
    vision INTEGER CHECK (vision BETWEEN 1 AND 100),
    composure INTEGER CHECK (composure BETWEEN 1 AND 100),
    
    -- Physical Attributes (1~100)
    strength INTEGER CHECK (strength BETWEEN 1 AND 100),
    stamina INTEGER CHECK (stamina BETWEEN 1 AND 100),

    created_at DATETIME DEFAULT CURRENT_TIMESTAMP
);

-- 조건 검색 속도 향상 인덱스

export interface Env {
  DB: D1Database;
}

export default {
  async fetch(request: Request, env: Env): Promise<Response> {
    const url = new URL(request.url);
    const path = url.pathname;

    // 1. 특정 능력치 조건 검색 API (예: /api/players?min_passing=80)
    if (path === "/api/players" && request.method === "GET") {
      const minPassing = url.searchParams.get("min_passing") || "1";
      const minFinishing = url.searchParams.get("min_finishing") || "1";

      const { results } = await env.DB.prepare(
        `SELECT id, name_ko, position_main, team, current_ability, passing, finishing 
         FROM players 
         WHERE passing >= ? AND finishing >= ? 
         ORDER BY current_ability DESC`
      )
        .bind(parseInt(minPassing), parseInt(minFinishing))
        .all();

      return Response.json({ success: true, count: results.length, data: results });
    }

    // 2. 선수 상세 조회 API (FM 프로필 객체 반환) (/api/players/:id)
    if (path.startsWith("/api/players/") && request.method === "GET") {
      const id = path.split("/")[3];
      const player: any = await env.DB.prepare("SELECT * FROM players WHERE id = ?")
        .bind(id)
        .first();

      if (!player) {
        return Response.json({ success: false, message: "Player not found" }, { status: 404 });
      }

      // FM 프로필 화면용 계층구조 객체 재구성
      const fmProfileResponse = {
        id: player.id,
        name_ko: player.name_ko,
        name_en: player.name_en,
        team: player.team,
        position: player.position_main,
        current_ability: player.current_ability,
        attributes: {
          technical: {
            finishing: player.finishing,
            passing: player.passing,
            tackling: player.tackling,
            technique: player.technique,
          },
          mental: {
            vision: player.vision,
            composure: player.composure,
          },
          physical: {
            strength: player.strength,
            stamina: player.stamina,
          },
        },
      };

      return Response.json({ success: true, data: fmProfileResponse });
    }

    return Response.json({ message: "K-League FM API Server" });
  },
};
